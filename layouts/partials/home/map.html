{{- /* The recorded network as an SVG, drawn at build time from the same display lines the builder uses, over the Forest Service's wilderness boundary. The terrain shading underneath is USGS 3DEP, made by tools/relief.py. */ -}} {{- $f := partial "home/facts.html" . }} {{- $segs := (os.ReadFile "static/data/display.json" | transform.Unmarshal).segments }} {{- $wild := index (os.ReadFile "archive/usfs-wilderness-boundary.geojson" | transform.Unmarshal).features 0 }} {{- $rings := $wild.geometry.coordinates }} {{- $minLat := 90.0 }}{{ $maxLat := -90.0 }}{{ $minLon := 180.0 }}{{ $maxLon := -180.0 }} {{- range $segs }}{{ range . }} {{- $minLat = math.Min $minLat (index . 0) }}{{ $maxLat = math.Max $maxLat (index . 0) }} {{- $minLon = math.Min $minLon (index . 1) }}{{ $maxLon = math.Max $maxLon (index . 1) }} {{- end }}{{ end }} {{- range $rings }}{{ range . }}{{/* GeoJSON is lon,lat */}} {{- $minLat = math.Min $minLat (index . 1) }}{{ $maxLat = math.Max $maxLat (index . 1) }} {{- $minLon = math.Min $minLon (index . 0) }}{{ $maxLon = math.Max $maxLon (index . 0) }} {{- end }}{{ end }} {{- $k := 0.829 }}{{/* cos 34°: keeps a mile east the same length as a mile north */}} {{- $w := 1000.0 }}{{ $s := div $w (mul (sub $maxLon $minLon) $k) }} {{- $h := math.Ceil (mul (sub $maxLat $minLat) $s) }} {{- $r := os.ReadFile "static/img/relief.json" | transform.Unmarshal }} {{- $d := slice }} {{- range $rings }} {{- $xy := slice }} {{- range . }}{{ $xy = $xy | append (printf "%.1f,%.1f" (mul (mul (sub (index . 0) $minLon) $k) $s) (mul (sub $maxLat (index . 1)) $s)) }}{{ end }} {{- $d = $d | append (printf "M%sZ" (delimit $xy "L")) }} {{- end }} {{- $byTrail := dict }} {{- range $id, $pts := $segs }} {{- $t := (index $f.seg $id).t }} {{- $xy := slice }} {{- range $pts }}{{ $xy = $xy | append (printf "%.1f,%.1f" (mul (mul (sub (index . 1) $minLon) $k) $s) (mul (sub $maxLat (index . 0)) $s)) }}{{ end }} {{- $line := printf `` (delimit $xy " ") }} {{- $byTrail = merge $byTrail (dict $t.slug (add (index $byTrail $t.slug | default "") $line)) }} {{- end }}
