{{- /* Figures the landing page reads, all from the built data: the trail count (roads not counted), miles of everything recorded, the newest note's date, and a segment id -> trail lookup for the map. */ -}} {{- $seg := dict }} {{- $trails := 0 }}{{ $miles := 0.0 }} {{- range $t := site.Data.trails }} {{- if ne $t.kind "road" }}{{ $trails = add $trails 1 }}{{ end }} {{- $miles = add $miles $t.miles }} {{- range $t.legs }}{{ $seg = merge $seg (dict .id (dict "t" $t "l" .)) }}{{ end }} {{- end }} {{- $last := "0000" }} {{- range site.Data.conditions }}{{ range . }}{{ if gt .date $last }}{{ $last = .date }}{{ end }}{{ end }}{{ end }} {{- return dict "trails" $trails "miles" $miles "last" $last "seg" $seg }}
